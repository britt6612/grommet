🎯 WHAT IS VANILLA EXTRACT?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  Zero-runtime, type-safe CSS-in-TypeScript
  
  ✅ Write styles in TypeScript (.css.ts files)
  ✅ Full type safety for themes & variants  
  ✅ Zero runtime overhead (build-time extraction)
  ✅ Excellent IDE support with autocomplete
  ✅ Framework agnostic (works everywhere)

�� KEY CONCEPTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

     - Uses recipe() for className
     - Type-safe variants

💡 CODE EXAMPLE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  // theme.contract.css.ts
  export const themeVars = createThemeContract({
    button: { padding: { vertical: null } }
  });
  
  // grommet.theme.css.ts
  export const lightTheme = createTheme(themeVars, {
    button: { padding: { vertical: '4px' } }
  });
  
  // button.css.ts
  export const buttonRecipe = recipe({
    base: { padding: themeVars.button.padding.vertical },
    variants: {
      kind: {
        primary: { background: '#7D4CDB' },
      },
    },
  });
  
  // Button.tsx
  const Button = ({ kind }) => (
    <button className={buttonRecipe({ kind })} />
  );
